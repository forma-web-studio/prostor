export interface LayoutVariant {
  id: LayoutId;
  title: string;
  advantage: string;
  tradeoff: string;
  plan: MediaRef;
  visual: MediaRef;
  alt: string;

  spatialRef: string;
  action: Action;
}
export interface Project {
  id: ProjectId;
  slug: string;
  title: string;

  brief: string;
  question: string;
  decision: string;
  tradeoff: string;
  cover: MediaRef;
  relatedServices: readonly ServiceId[];
  action: Action;
  next: Action;
}
